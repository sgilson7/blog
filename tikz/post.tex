% A uniform bounding box, so every figure has the same natural width and the
% column scales them all by the same factor -- otherwise a narrow figure is
% upscaled more than a wide one and their labels end up different sizes.

      rectangle ([xshift=\HalfW]current bounding box.center);
\end{tikzpicture}
